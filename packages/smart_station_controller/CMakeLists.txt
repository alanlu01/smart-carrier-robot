cmake_minimum_required(VERSION 3.8)
project(smart_station_controller)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(smart_navigation_policy_bt_node SHARED src/navigation_policy.cpp)
target_compile_definitions(smart_navigation_policy_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(smart_navigation_policy_bt_node rclcpp std_msgs geometry_msgs behaviortree_cpp)
target_link_libraries(smart_navigation_policy_bt_node nlohmann_json::nlohmann_json)
install(TARGETS smart_navigation_policy_bt_node LIBRARY DESTINATION lib)
add_library(station_controller SHARED src/station_controller.cpp)
target_include_directories(station_controller PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(station_controller nav2_core nav2_mppi_controller
  nav2_costmap_2d pluginlib rclcpp rclcpp_lifecycle tf2_geometry_msgs std_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS station_controller ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(speed_envelope_test test/speed_envelope_test.cpp)
  target_include_directories(speed_envelope_test PRIVATE include)
  add_test(NAME station_speed_envelope COMMAND speed_envelope_test)
endif()
ament_export_libraries(station_controller)
ament_export_include_directories(include)
ament_export_dependencies(nav2_mppi_controller nav2_core)
ament_package()
